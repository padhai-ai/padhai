export default { async fetch(request, env) { const url = new URL(request.url); // ========================================== // COMMON HELPERS // ========================================== function json(data, status = 200) { return new Response(JSON.stringify(data), { status, headers: { "Content-Type": "application/json" } }); } async function hashPassword(password) { const buffer = await crypto.subtle.digest( "SHA-256", new TextEncoder().encode(password) ); return Array.from(new Uint8Array(buffer)) .map(byte => byte.toString(16).padStart(2, "0")) .join(""); } async function runAI(prompt, system = "") { if (!env.AI) { throw new Error("Workers AI binding is missing."); } const result = await env.AI.run( "@cf/meta/llama-3.2-3b-instruct", { messages: [ { role: "system", content: system || "You are PadhAI, a friendly educational AI tutor." }, { role: "user", content: prompt } ], max_tokens: 1500, temperature: 0.4 } ); return result?.response || ""; } function cleanJSON(text) { if (typeof text !== "string") { return text; } let cleaned = text.trim(); cleaned = cleaned .replace(/```json/gi, "") .replace(/```/g, "") .trim(); const first = cleaned.indexOf("{"); const last = cleaned.lastIndexOf("}"); if (first !== -1 && last !== -1) { cleaned = cleaned.slice(first, last + 1); } return JSON.parse(cleaned); } // ========================================== // SIGNUP // ========================================== if ( url.pathname === "/api/signup" && request.method === "POST" ) { try { const body = await request.json(); const name = String(body.name || "").trim(); const email = String(body.email || "") .trim() .toLowerCase(); const password = String(body.password || ""); if (!name || !email || !password) { return json({ success: false, error: "All fields are required." }, 400); } if (password.length < 6) { return json({ success: false, error: "Password must be at least 6 characters." }, 400); } const passwordHash = await hashPassword(password); const result = await env.DB .prepare(` INSERT INTO users (name, email, password_hash) VALUES (?, ?, ?) `) .bind( name, email, passwordHash ) .run(); return json({ success: true, id: result.meta?.last_row_id || null, name, email }); } catch (error) { console.error( "SIGNUP ERROR", error ); if ( String(error.message || "") .toLowerCase() .includes("unique") ) { return json({ success: false, error: "Email already registered." }, 409); } return json({ success: false, error: "Could not create account." }, 500); } } // ========================================== // LOGIN // ========================================== if ( url.pathname === "/api/login" && request.method === "POST" ) { try { const body = await request.json(); const email = String(body.email || "") .trim() .toLowerCase(); const password = String(body.password || ""); if (!email || !password) { return json({ success: false, error: "Email and password are required." }, 400); } const passwordHash = await hashPassword(password); const user = await env.DB .prepare(` SELECT id, name, email FROM users WHERE email = ? AND password_hash = ? `) .bind( email, passwordHash ) .first(); if (!user) { return json({ success: false, error: "Invalid email or password." }, 401); } return json({ success: true, id: user.id, name: user.name, email: user.email }); } catch (error) { console.error( "LOGIN ERROR", error ); return json({ success: false, error: "Login failed." }, 500); } } // ========================================== // AI TUTOR // ========================================== if ( url.pathname === "/api/tutor" && request.method === "POST" ) { try { const body = await request.json(); const question = String(body.question || "").trim(); if (!question) { return json({ success: false, error: "Please enter a question." }, 400); } const answer = await runAI( question, `You are PadhAI Tutor. Help students understand subjects clearly. Rules: - Explain in simple language. - Give step-by-step explanations. - Give examples when useful. - Be friendly. - Do not unnecessarily make answers complicated.` ); return json({ success: true, answer: answer || "I couldn't generate an answer." }); } catch (error) { console.error( "TUTOR ERROR", error ); return json({ success: false, error: "AI Tutor failed: " + String(error.message || "Unknown error") }, 500); } } // ========================================== // STUDY // ========================================== if ( url.pathname === "/api/study" && request.method === "POST" ) { try { const body = await request.json(); const topic = String(body.topic || "").trim(); const level = String(body.level || "beginner").trim(); if (!topic) { return json({ success: false, error: "Please enter a study topic." }, 400); } const prompt = ` Create a study lesson about "${topic}". Student level: ${level} IMPORTANT: Return ONLY valid JSON. Do not use markdown. Do not put JSON inside code blocks. Use exactly this structure: { "title": "Lesson title", "introduction": "Short introduction", "explanation": "Detailed but easy explanation", "keyPoints": [ "Point 1", "Point 2", "Point 3", "Point 4" ], "example": "Simple example", "summary": "Short revision summary" } Make the lesson accurate and student-friendly. `; const raw = await runAI( prompt, "You are PadhAI Study Teacher. Always return valid JSON only." ); let study; try { study = cleanJSON(raw); } catch (error) { console.error( "STUDY JSON PARSE ERROR", raw ); return json({ success: false, error: "AI returned an invalid lesson. Please try again." }, 500); } if ( !study || typeof study !== "object" ) { return json({ success: false, error: "Invalid lesson received." }, 500); } const lesson = { title: String( study.title || topic ), introduction: String( study.introduction || "" ), explanation: String( study.explanation || "" ), keyPoints: Array.isArray(study.keyPoints) ? study.keyPoints .map(x => String(x)) .filter(Boolean) : [], example: String( study.example || "" ), summary: String( study.summary || "" ) }; if ( !lesson.introduction || !lesson.explanation || !lesson.summary ) { return json({ success: false, error: "The lesson was incomplete. Please try again." }, 500); } return json({ success: true, topic, level, // IMPORTANT: // HTML can use either data.lesson // or the fields directly. lesson, title: lesson.title, introduction: lesson.introduction, explanation: lesson.explanation, keyPoints: lesson.keyPoints, example: lesson.example, summary: lesson.summary }); } catch (error) { console.error( "STUDY ERROR", error ); return json({ success: false, error: "Study generation failed: " + String(error.message || "Unknown error") }, 500); } } // ========================================== // AI QUIZ // ========================================== if ( url.pathname === "/api/quiz" && request.method === "POST" ) { try { const body = await request.json(); const topic = String(body.topic || "").trim(); let count = Number(body.count) || 5; if (count < 1) count = 1; if (count > 10) count = 10; if (!topic) { return json({ success: false, error: "Please enter a quiz topic." }, 400); } const prompt = ` Create exactly ${count} multiple-choice questions about "${topic}". Return ONLY valid JSON. Use exactly this format: { "questions": [ { "question": "Question text", "options": [ "Option A", "Option B", "Option C", "Option D" ], "answer": 0 } ] } Rules: - Exactly ${count} questions. - Every question must have exactly 4 options. - "answer" must be 0, 1, 2, or 3. - The answer number represents the correct option. - Questions must be educational and accurate. - Do not use markdown. - Do not add anything outside the JSON. `; const raw = await runAI( prompt, "You are PadhAI Quiz Generator. Always return valid JSON only." ); let parsed; try { parsed = cleanJSON(raw); } catch (error) { console.error( "QUIZ JSON PARSE ERROR", raw ); return json({ success: false, error: "AI returned an invalid quiz. Please try again." }, 500); } let questions = Array.isArray(parsed) ? parsed : parsed?.questions; if (!Array.isArray(questions)) { return json({ success: false, error: "No quiz questions were received." }, 500); } const validQuestions = []; for (const item of questions) { if (!item) continue; const question = String( item.question || "" ).trim(); if (!question) continue; if ( !Array.isArray(item.options) || item.options.length < 4 ) { continue; } const options = item.options .slice(0, 4) .map(x => String(x || "").trim() ); if ( options.length !== 4 || options.some(x => !x) ) { continue; } const answer = Number(item.answer); if ( !Number.isInteger(answer) || answer < 0 || answer > 3 ) { continue; } validQuestions.push({ question, options, answer }); if ( validQuestions.length >= count ) { break; } } if (!validQuestions.length) { return json({ success: false, error: "PadhAI could not create valid questions. Please try again." }, 500); } return json({ success: true, topic, questions: validQuestions }); } catch (error) { console.error( "QUIZ ERROR", error ); return json({ success: false, error: "Quiz generation failed: " + String(error.message || "Unknown error") }, 500); } } // ========================================== // SAVE QUIZ SCORE // ========================================== if ( url.pathname === "/api/quiz-score" && request.method === "POST" ) { try { const body = await request.json(); const userId = Number(body.userId); const topic = String(body.topic || "").trim(); const score = Number(body.score); const total = Number(body.total); if ( !userId || !topic || !Number.isFinite(score) || !Number.isFinite(total) ) { return json({ success: false, error: "Invalid quiz score data." }, 400); } await env.DB .prepare(` CREATE TABLE IF NOT EXISTS quiz_scores ( id INTEGER PRIMARY KEY AUTOINCREMENT, user_id INTEGER NOT NULL, topic TEXT NOT NULL, score INTEGER NOT NULL, total INTEGER NOT NULL, created_at TEXT DEFAULT CURRENT_TIMESTAMP ) `) .run(); await env.DB .prepare(` INSERT INTO quiz_scores (user_id, topic, score, total) VALUES (?, ?, ?, ?) `) .bind( userId, topic, score, total ) .run(); return json({ success: true, message: "Quiz score saved." }); } catch (error) { console.error( "SCORE ERROR", error ); return json({ success: false, error: "Could not save quiz score." }, 500); } } // ========================================== // QUIZ HISTORY // ========================================== if ( url.pathname === "/api/quiz-history" && request.method === "GET" ) { try { const userId = Number( url.searchParams.get("userId") ); if (!userId) { return json({ success: false, error: "User ID is required." }, 400); } await env.DB .prepare(` CREATE TABLE IF NOT EXISTS quiz_scores ( id INTEGER PRIMARY KEY AUTOINCREMENT, user_id INTEGER NOT NULL, topic TEXT NOT NULL, score INTEGER NOT NULL, total INTEGER NOT NULL, created_at TEXT DEFAULT CURRENT_TIMESTAMP ) `) .run(); const result = await env.DB .prepare(` SELECT id, topic, score, total, created_at FROM quiz_scores WHERE user_id = ? ORDER BY id DESC LIMIT 20 `) .bind(userId) .all(); return json({ success: true, scores: result.results || [] }); } catch (error) { console.error( "HISTORY ERROR", error ); return json({ success: false, error: "Could not load quiz history." }, 500); } } // ========================================== // DATABASE TEST // ========================================== if ( url.pathname === "/api/test-db" ) { try { const result = await env.DB .prepare(` SELECT name FROM sqlite_master WHERE type='table' ORDER BY name `) .all(); return json({ success: true, tables: result.results || [] }); } catch (error) { return json({ success: false, error: String(error.message || error) }, 500); } } // ========================================== // AI TEST // ========================================== if ( url.pathname === "/api/test-ai" ) { try { if (!env.AI) { return json({ success: false, error: "AI binding is missing." }, 500); } const result = await env.AI.run( "@cf/meta/llama-3.2-3b-instruct", { messages: [ { role: "user", content: "Say exactly: PadhAI AI is working!" } ] } ); return json({ success: true, response: result?.response || "" }); } catch (error) { console.error( "AI TEST ERROR", error ); return json({ success: false, error: String(error.message || error) }, 500); } } // ========================================== // WEBSITE // ========================================== if (env.ASSETS) { return env.ASSETS.fetch( request ); } // ========================================== // FALLBACK // ========================================== return new Response( "PadhAI Worker is running.", { status: 200, headers: { "content-type": "text/plain;charset=UTF-8" } } ); } };
